```ts
export interface SnapshotRow {
  reference_date: string;        // YYYY-MM-DD
  declared_total_cents: number;
  import_id: number;
  imported_at: string;           // YYYY-MM-DD HH:MM:SS
}

/**
 * Returns XPerformance imports as valuation snapshots, sorted by
 * reference_date ASC then imported_at DESC (defensive — if two imports
 * share a reference_date, the most recently imported wins).
 *
 * Filters require declared_total_cents AND reference_date to be NOT
 * NULL — other source_types are excluded.
 *
 * `periodFrom` and `periodTo` are inclusive ISO date bounds on
 * reference_date.
 */
export function listValuationSnapshots(
  db: Database,
  periodFrom?: string,
  periodTo?: string,
): SnapshotRow[] {
  const clauses: string[] = [
    "source_type = 'pdf_xperformance'",
    "declared_total_cents IS NOT NULL",
    "reference_date IS NOT NULL",
  ];
  const params: unknown[] = [];
  if (periodFrom) {
    clauses.push("reference_date >= ?");
    params.push(periodFrom);
  }
  if (periodTo) {
    clauses.push("reference_date <= ?");
    params.push(periodTo);
  }
  const sql = `
    SELECT reference_date, declared_total_cents, id AS import_id, imported_at
      FROM imports
     WHERE ${clauses.join(" AND ")}
     ORDER BY reference_date ASC, imported_at DESC, id DESC`;
  return db.prepare(sql).all(...params) as SnapshotRow[];
}
```

```ts
  cash_flows_used: number;

/**
 * Caller MUST validate snapshots.length >= 2 before calling.
 * Snapshots MUST be sorted by reference_date ASC.
 * cashFlows are the full set from listCashFlows; the service filters
 * to the snapshot window.
 *
 * Returns a unified ReturnsCore — wrappers project to TwrResult or
 * MwrResult.

export function computeReturns(
  snapshots: readonly SnapshotRow[],
  cashFlows: readonly CashFlowRow[],
): ReturnsCore {
  // 1) dedupe same-date snapshots (already done by repo's
  //    `imported_at DESC` tiebreaker, but defensive)
  // 2) filter cash flows to (snapshots[0].reference_date, snapshots[N-1].reference_date]
  //    inclusive on both ends — same-day flows on first snapshot land
  //    in first sub-period; same-day on last go in last sub-period.
  // 3) compute TWR via Modified Dietz chain
  // 4) compute MWR via bisection IRR
  // 5) compute annualization, days
  // 6) warnings: sparse history, large flow, tiny period, mwr non-convergence
  //    — stale-tail and filtered-flows warnings are emitted by the
  //    TOOL WRAPPER, not the service (the service doesn't know about
  //    `now` or about flows excluded before the call).
}

export function computeTwr(
  snapshots: readonly SnapshotRow[],
  cashFlows: readonly CashFlowRow[],
): TwrCore {
  return computeReturns(snapshots, cashFlows).twr;
}

export function computeMwr(
  snapshots: readonly SnapshotRow[],
  cashFlows: readonly CashFlowRow[],
): MwrCore {
  return computeReturns(snapshots, cashFlows).mwr;
}
```

### Date arithmetic

ISO dates compare lexicographically — same trick as v0.8. For
`days_between(a, b)` use:

```ts
function daysBetween(a: string, b: string): number {
  // Both YYYY-MM-DD. Use Date.UTC parse → ms → divide. Pure date math,
  // no timezone surprise because the local midnight assumption is
  // bypassed via Date.UTC.
  const aMs = Date.UTC(
    parseInt(a.slice(0, 4), 10),
    parseInt(a.slice(5, 7), 10) - 1,
    parseInt(a.slice(8, 10), 10),
  );
  const bMs = Date.UTC(
    parseInt(b.slice(0, 4), 10),
    parseInt(b.slice(5, 7), 10) - 1,
    parseInt(b.slice(8, 10), 10),
  );
  return Math.round((bMs - aMs) / 86_400_000);
}
```

This helper goes in `services/returns.ts`, not exported (unless test
file imports it — then export).

### Modified Dietz formula (one sub-period)

```ts
function modifiedDietz(
  vStart: number,
  vEnd: number,
  flowsInPeriod: Array<{ amount: number; daysFromStart: number }>,
  periodDays: number,
): number {
  const netCf = flowsInPeriod.reduce((s, f) => s + f.amount, 0);
  const weightedCf = flowsInPeriod.reduce(
    (s, f) => s + f.amount * ((periodDays - f.daysFromStart) / periodDays),
    0,
  );
  const numerator = vEnd - vStart - netCf;
  const denominator = vStart + weightedCf;
  if (denominator === 0) return 0;
  return numerator / denominator;
}
```

`amount` is signed (APORTE → +, RESGATE → -).
`daysFromStart` is days from sub-period start to the flow date.
A flow on the same day as the start has `daysFromStart = 0` and so
`weight = 1` (fully present for the whole period).
A flow on the same day as the end has `daysFromStart = periodDays` and
`weight = 0` (effectively no time to earn).

### Bisection IRR

```ts
function npv(rate: number, flows: Array<{ date: string; amount: number }>, anchorDate: string): number {
  return flows.reduce((sum, f) => {
    const t = daysBetween(anchorDate, f.date) / 365;
    return sum + f.amount / Math.pow(1 + rate, t);
  }, 0);
}

function bisectIrr(
  flows: Array<{ date: string; amount: number }>,
  anchorDate: string,
): { rate: number | null; converged: boolean; iterations: number; npvAtSolution: number | null } {
  let lo = -0.99;
  let hi = 10.0;
  const fLo = npv(lo, flows, anchorDate);
  const fHi = npv(hi, flows, anchorDate);
  // Need sign change. If not present, no IRR in range.
  if (fLo * fHi > 0) {
    return { rate: null, converged: false, iterations: 0, npvAtSolution: null };
  }
  const TOL = 1e-6;
  for (let i = 1; i <= 100; i++) {
    const mid = (lo + hi) / 2;
    const fMid = npv(mid, flows, anchorDate);
    if (Math.abs(fMid) < TOL) {
      return { rate: mid, converged: true, iterations: i, npvAtSolution: fMid };
    }
    if (fLo * fMid < 0) {
      hi = mid;
    } else {
      lo = mid;
    }
  }
  // Did not converge to tolerance in 100 iterations.
  return { rate: null, converged: false, iterations: 100, npvAtSolution: null };
}
```

`rate` here is **annualized** — note the `t = days / 365` in NPV. So
`mwr_annualized = rate` directly; `mwr_period = (1 + rate)^(days/365) - 1`
gives the period return.

**IMPORTANT signed-flow convention for MWR:**

- For each `APORTE` cash flow: `amount = -amount_brl` (investor pays)
- For each `RESGATE` cash flow: `amount = +amount_brl` (investor receives)
- Synthetic initial NAV: `amount = -vFirst` at `snapshots[0].reference_date`
- Synthetic terminal NAV: `amount = +vLast` at `snapshots[last].reference_date`

Result: typical pattern is initial -V_first, mid-period -aporte values,
and +V_last terminal. Sign change exists → IRR has a solution.

**Hand-computed reference values:**

For the canonical CFA Modified Dietz example:

- V_start = 100,000 on 2025-01-01
- Aporte 10,000 on 2025-01-16 (mid-month)
- V_end = 115,000 on 2025-02-01 (31 days later)
- Net CF = +10,000; weighted = 10,000 × (31-15)/31 = 10,000 × 0.516 = 5,161.29
- Numerator = 115,000 - 100,000 - 10,000 = 5,000
- Denominator = 100,000 + 5,161.29 = 105,161.29
- Period return = 5,000 / 105,161.29 ≈ 0.04755 (4.755%)

For MWR with no cash flows:

- V_start = 100,000 on 2025-01-01
- V_end = 110,000 on 2026-01-01 (365 days, exactly 1 year)
- Cash flows: [(2025-01-01, -100,000), (2026-01-01, +110,000)]
- IRR satisfies: -100,000 + 110,000 / (1+r)^1 = 0 → r = 0.10 exactly
- `mwr_annualized = 0.10`, `mwr_period = 0.10` (since period = 1 year)

These two cases are the foundation — if the implementation gets them
both right to 4 decimals, the rest is sign and edge handling.

Write at least one test per case in spec § 7.1 (17 listed). Add
`mwr_period` reconstruction test: `(1 + mwr_annualized)^(days/365) - 1`
should equal `mwr_period`.

```ts
import { z } from "zod";
import type { Database } from "better-sqlite3";
import { getDb } from "../storage/db.js";
import {
  listValuationSnapshots,
  type SnapshotRow,
} from "../storage/imports-repo.js";
import { listCashFlowsSince } from "../storage/cash-flows-repo.js";
import { computeTwr, type TwrSubPeriod } from "../services/returns.js";

export const calculateTwrSchema = z.object({
  period_from: z.string().regex(/^\d{4}-\d{2}-\d{2}$/).optional(),
  period_to: z.string().regex(/^\d{4}-\d{2}-\d{2}$/).optional(),
});
export type CalculateTwrInput = z.infer<typeof calculateTwrSchema>;

export interface TwrResult {
  period_from: string;
  period_to: string;
  days: number;
  snapshots_used: number;
  cash_flows_used: number;
  twr_period: number;
  twr_annualized: number | null;
  sub_periods: TwrSubPeriod[];
  warnings: string[];
}

export interface CalculateTwrDeps {
  db?: Database;
  now?: Date;
}

export class InsufficientHistoryError extends Error {
  constructor(found: number) {
    super(
      `Need at least 2 XPerformance imports to compute returns. Found ${found}. ` +
        `Import another XPerformance PDF (or pass an earlier period_from / later period_to that covers ≥ 2 imports).`,
    );
    this.name = "InsufficientHistoryError";
  }
}

export async function calculateTwr(
  input: CalculateTwrInput,
  deps: CalculateTwrDeps = {},
): Promise<TwrResult> {
  const db = deps.db ?? getDb();
  const now = deps.now ?? new Date();

  const snapshots = listValuationSnapshots(db, input.period_from, input.period_to);
  if (snapshots.length < 2) {
    throw new InsufficientHistoryError(snapshots.length);
  }

  // Load all cash flows since the first snapshot's date; service filters
  // to the window. (We use listCashFlowsSince — already exists from v0.8 —
  // rather than re-implementing.)
  const cashFlows = listCashFlowsSince(db, snapshots[0]!.reference_date);

  const core = computeTwr(snapshots, cashFlows);

  // Tool-layer warnings (computed here, not in the pure service):
  const warnings = [...core.warnings];
  const lastDate = snapshots[snapshots.length - 1]!.reference_date;
  const nowDate = now.toISOString().slice(0, 10);
  const daysSinceLast = daysBetween(lastDate, nowDate);
  if (input.period_to == null && daysSinceLast > 60) {
    warnings.push(
      `Last snapshot is from ${lastDate}; latest portfolio state may be significantly different. Import a fresh XPerformance to refresh.`,
    );
  }

  return {
    period_from: snapshots[0]!.reference_date,
    period_to: lastDate,
    days: daysBetween(snapshots[0]!.reference_date, lastDate),
    snapshots_used: snapshots.length,
    cash_flows_used: /* counted by the service via core ... */ 0, // see note below
    twr_period: core.twr_period,
    twr_annualized: core.twr_annualized,
    sub_periods: core.sub_periods,
    warnings,
  };
}

// Local daysBetween helper, OR import from services/returns.ts (export it).
```

### Tool descriptions (for the TOOLS registry)

```text
calculate_twr:
  Time-weighted return (TWR) over the XPerformance import history.
  Removes the effect of when you contributed/withdrew — pure portfolio
  performance metric, GIPS-compliant. Uses Modified Dietz between
  consecutive imports and chains sub-periods geometrically. Requires
  ≥ 2 XPerformance imports. Returns period_return, annualized return,
  per-sub-period breakdown, and quality warnings (sparse history,
  large cash flows, stale tail).

calculate_mwr:
  Money-weighted return (MWR / IRR) — the rate that reflects when
  you put money in vs. out. Computed via bisection over signed cash
  flows including synthetic initial/terminal NAV. Requires ≥ 2
  XPerformance imports. Returns mwr_period, mwr_annualized,
  cash_flow_breakdown, convergence diagnostics, and warnings.
```
